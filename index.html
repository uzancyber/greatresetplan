api/verify-auth.js (Node.js Serverless Function Vercel)

// Letakkan file ini di: /api/verify-auth.js pada repository GitHub Anda.
// Vercel otomatis menjalankan kode ini di SERVER (pengunjung tidak bisa melihat daftar email ini).

const WHITELISTED_EMAILS = [
  "uz.research@gmail.com",
  "member.vip@gmail.com",
  "admin.uz@gmail.com"
  // Tambahkan alamat Gmail anggota resmi Anda di sini, atau simpan di Environment Variable Vercel: WHITELIST_EMAILS
];

export default async function handler(req, res) {
  if (req.method !== "POST") {
    return res.status(405).json({ allowed: false, error: "Method not allowed" });
  }

  try {
    const { accessToken, idToken } = req.body || {};
    let googleUser = null;

    if (accessToken) {
      const googleRes = await fetch("https://www.googleapis.com/oauth2/v3/userinfo", {
        headers: { Authorization: `Bearer ${accessToken}` }
      });
      if (googleRes.ok) googleUser = await googleRes.json();
    } else if (idToken) {
      const tokenRes = await fetch(`https://oauth2.googleapis.com/tokeninfo?id_token=${encodeURIComponent(idToken)}`);
      if (tokenRes.ok) googleUser = await tokenRes.json();
    }

    if (!googleUser || !googleUser.email || googleUser.email_verified === false || googleUser.email_verified === "false") {
      return res.status(401).json({ allowed: false, error: "Token Google tidak valid." });
    }

    const envEmails = process.env.WHITELIST_EMAILS
      ? process.env.WHITELIST_EMAILS.split(",").map(e => e.trim().toLowerCase())
      : [];
    const masterList = [...WHITELISTED_EMAILS.map(e => e.trim().toLowerCase()), ...envEmails];

    const email = googleUser.email.trim().toLowerCase();
    const isAllowed = masterList.includes(email);

    return res.status(200).json({
      allowed: isAllowed,
      email: email,
      name: googleUser.name || "",
      picture: googleUser.picture || ""
    });
  } catch (err) {
    return res.status(500).json({ allowed: false, error: "Server verification error" });
  }
}
